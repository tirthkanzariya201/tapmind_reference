export const LatestBadge = ({ platform, mediation, builtOn }) => {
  const version = typeof builtOn === "string" ? builtOn.trim() : "";
  if (!version) return null;
  const current = NATIVE_CURRENT[`${platform}|${mediation}`];
  if (!current) return null;
  const normalize = (v) => v.replace(/^v/i, "");
  if (normalize(version) !== normalize(current)) return null;
  return (
    <Badge color="green" size="sm" shape="pill">
      Latest
    </Badge>
  );
};
